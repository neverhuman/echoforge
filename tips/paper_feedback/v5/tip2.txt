I studied the current uploaded PDF and the repo paper path. The problem is not one bad plot. It is a **paper-figure system problem**: too much text inside figures, manual positioning, inconsistent table/plot styles, no shared paper style, overloaded multi-panel figures, and a few real code bugs. The worst current failures are visible in Fig. 1 page 2, Fig. 3 page 5, Fig. 4 page 6, Fig. 5 page 7, Fig. 6 page 8/9, and Fig. 7/8 page 9/10. Fig. 6 is especially damaging: the radar model-card figure has overlapping cue-definition text, a cramped table, and in one PDF version the “Chirps” column is rendered as `0`, which directly undermines the model-card credibility. 

The paper’s **claim boundary is good** and the abstract is basically defensible: it clearly states strict-open, public-proxy, non-measured-truth framing, the 10,000 groups / 50 positives / 30,000 records setup, the 4,500-record holdout, and the locked candidate’s 0.833 point Recall@≤1%FPR and 0.699 LCB95 versus 0.292 / 0.083 for prior fusion.  But the figures currently make the work look less mature than the text. A tier-1 version needs the figures to feel like a controlled evidence system, not generated dashboards.

Below is a concrete, agent-ready patch plan. The highest-impact implementation is: **replace the figure-rendering style layer, rewrite Figs. 1–8 in a common IEEE style, generate vector PDFs for all non-heatmap figures, and update the LaTeX placement for Fig. 8.**

```python
def _wrap_label(value: Any, width: int = 18) -> str:
    text = str(value).replace("_", " ").replace("-", "-").strip()
    return "\n".join(
        textwrap.wrap(
            text,
            width=width,
            break_long_words=False,
            break_on_hyphens=False,
        )
    )


def _method_label(method: str, width: int = 18) -> str:
    label = METHOD_LABELS.get(method, method.replace("_", " "))
    return _wrap_label(label, width=width)


def _family_label(family: str, width: int = 18) -> str:
    label = FAMILY_LABELS.get(family, family.replace("_", " "))
    return _wrap_label(label, width=width)


def _style_axes(ax: Any, *, grid: str = "x") -> None:
    if grid in ("x", "both"):
        ax.xaxis.grid(True, color=GRID, linewidth=GRID_WIDTH)
    if grid in ("y", "both"):
        ax.yaxis.grid(True, color=GRID, linewidth=GRID_WIDTH)
    ax.set_axisbelow(True)
    for spine in ("top", "right"):
        ax.spines[spine].set_visible(False)
    ax.spines["left"].set_color(GRID)
    ax.spines["bottom"].set_color(GRID)
    ax.spines["left"].set_linewidth(LINE_WIDTH)
    ax.spines["bottom"].set_linewidth(LINE_WIDTH)


def _style_count_axis(ax: Any) -> None:
    _style_axes(ax, grid="x")
    ax.xaxis.set_major_locator(MaxNLocator(nbins=4, integer=True))
    ax.tick_params(axis="both", length=2.2, width=0.6, pad=1.5)


def _top_items_with_other(
    counts: dict[str, Any],
    *,
    max_items: int = 7,
) -> list[tuple[str, int]]:
    cleaned = {str(k): int(v) for k, v in counts.items()}
    ordered = sorted(cleaned.items(), key=lambda item: item[1], reverse=True)
    if len(ordered) <= max_items:
        return ordered
    kept = ordered[: max_items - 1]
    kept.append(("other", sum(value for _name, value in ordered[max_items - 1 :])))
    return kept


def _add_panel_letter(ax: Any, letter: str) -> None:
    ax.text(
        -0.08,
        1.04,
        letter,
        transform=ax.transAxes,
        fontsize=8.8,
        weight="bold",
        va="bottom",
        ha="left",
        color=INK,
    )


def _barh_value_labels(
    ax: Any,
    values: list[float],
    *,
    fmt: str = "{:.3f}",
    pad_fraction: float = 0.018,
    fontsize: float = ANNOTATION_SIZE,
) -> None:
    finite = [value for value in values if math.isfinite(value)]
    xmax = max(finite) if finite else 1.0
    for idx, value in enumerate(values):
        if not math.isfinite(value):
            continue
        ax.text(
            value + xmax * pad_fraction,
            idx,
            fmt.format(value),
            va="center",
            ha="left",
            fontsize=fontsize,
            color=MUTED,
        )


def _fig_note(fig: Any, text: str, *, y: float = 0.012) -> None:
    fig.text(
        0.015,
        y,
        text,
        ha="left",
        va="bottom",
        fontsize=CAPTION_NOTE_SIZE,
        color=MUTED,
    )


def _nonempty_calibration_bins(selected_row: dict[str, Any], evaluation: dict[str, Any]) -> list[dict[str, Any]]:
    bins = selected_row.get("calibration_bins", [])
    if not bins:
        bins = evaluation.get("calibration", {}).get("selected", {}).get("bins", [])
    output = []
    for row in bins:
        count = _safe_float(row.get("count"))
        mean_score = _safe_float(row.get("mean_score"))
        empirical = _safe_float(row.get("empirical_positive_rate"))
        if count > 0 and math.isfinite(mean_score) and math.isfinite(empirical):
            output.append(row)
    return output
```

Replace `_save_figure()` with:

```python
def _save_figure(fig: Any, filename: str) -> Path:
    FIGURE_DIR.mkdir(parents=True, exist_ok=True)
    png_path = FIGURE_DIR / filename

    metadata = {"Software": "EchoForge paper/generate_figures_major_upgrade_v2.py"}
    description = f"sources={_source_note(filename)}"
    if FIGURE_FALLBACK_NOTES.get(filename):
        description += f"; fallback={_fallback_note(filename)}"
    metadata["Description"] = description

    # Draw first so constrained/tight layout has stable text extents.
    fig.canvas.draw_idle()

    fig.savefig(
        png_path,
        bbox_inches="tight",
        pad_inches=0.035,
        metadata=metadata,
    )

    # Keep range-Doppler heatmaps raster-only. All other paper figures should
    # also have vector PDFs so text, axes, and boxes remain sharp in IEEE PDF.
    if png_path.name != "iq_negative_samples.png":
        pdf_path = png_path.with_suffix(".pdf")
        fig.savefig(
            pdf_path,
            bbox_inches="tight",
            pad_inches=0.035,
            metadata=metadata,
        )

    plt.close(fig)
    return png_path
```

```python
def _synthetic_rd_proxy(group_id: str, phase: str, positive: bool) -> np.ndarray:
    if np is None:
        raise RuntimeError("numpy is required for synthetic range-Doppler fallback")

    rng = np.random.default_rng(_stable_seed("rd-fallback", group_id, phase))
    rows, cols = 24, 20
    yy = np.linspace(-1.0, 1.0, rows)[:, None]
    xx = np.linspace(0.0, 1.0, cols)[None, :]
    phase_idx = PHASES.index(phase)

    texture = 0.14 + 0.12 * rng.random((rows, cols))
    if positive:
        center = 0.27 + 0.19 * phase_idx
        ridge = np.exp(-((xx - center) ** 2) / (2.0 * 0.055**2))
        doppler = 0.48 + 0.35 * np.exp(-((yy - 0.18 * phase_idx) ** 2) / 0.16)
        signal = ridge * doppler
    else:
        center = 0.36 + 0.10 * ((stable_seed(group_id) % 5) / 4.0)
        smear = np.exp(-((xx - center) ** 2) / (2.0 * 0.17**2))
        doppler = 0.32 + 0.26 * np.sin(2.0 * math.pi * (yy + 0.18 * phase_idx)) ** 2
        signal = smear * doppler

    return _normalize01(texture + signal)
```

```latex
EchoForge is a strict-open synthetic benchmark for low-altitude uncrewed aerial system (UAS) radar and multimodal detection research. # 14. Add a figure quality gate
```

Add this to `paper/validate_paper.py` after `validate_figures()` or as a helper called inside it:

```python
def validate_figure_dimensions(figures_dir: Path) -> None:
    """Catch accidental tiny or blank paper figures before LaTeX build."""
    try:
        from PIL import Image
    except ImportError:
        return

    minimums = {
        "architecture_stack.png": (2400, 1000),
        "monte_carlo_split_flow.png": (2400, 1200),
        "kpi_ranking.png": (2400, 1200),
        "phase_kpi.png": (2400, 1100),
        "iq_drone_samples.png": (2400, 1200),
        "iq_negative_samples.png": (2400, 1200),
        "detector_ml_pipeline.png": (2400, 1100),
        "locked_algorithm.png": (2200, 900),
    }

    too_small: list[str] = []
    for name, (min_w, min_h) in minimums.items():
        path = figures_dir / name
        if not path.exists():
            continue
        with Image.open(path) as image:
            width, height = image.size
        if width < min_w or height < min_h:
            too_small.append(f"{name}={width}x{height}, expected at least {min_w}x{min_h}")

    if too_small:
        fail("paper figure(s) too small for IEEE layout: " + "; ".join(too_small))
```

Then in `main()` or wherever `validate_figures(figures_dir)` is called, call:

```python
validate_figure_dimensions(args.figures_dir)
```

This will not detect every overlap, but it prevents the agent from returning postage-stamp figures.
